\newcommand{\linkbutton}[1]{%
  \begingroup
  \setlength{\fboxsep}{2pt}%
  \setlength{\fboxrule}{0.35pt}%
  \fcolorbox{black!20}{black!6}{\footnotesize\sffamily #1}%
  \endgroup
}

% Field formats (explicit hyperlinks so buttons always click)
\DeclareFieldFormat{url}{\href{#1}{\linkbutton{Link}}}
\DeclareFieldFormat{doi}{\href{https://doi.org/#1}{\linkbutton{Link}}}
\DeclareFieldFormat{eprint:arxiv}{\href{https://arxiv.org/abs/#1}{\linkbutton{Link}}}

% usera holds an open-access PDF link
\DeclareFieldFormat{usera}{\href{#1}{\linkbutton{PDF}}}

\DeclareFieldFormat{urldate}{}

% Print exactly one primary link (DOI > URL > arXiv), plus optional PDF (usera),
% all right-aligned.
\renewbibmacro*{doi+eprint+url}{%
  \printtext{\hfill\mbox{%
    \iffieldundef{doi}
      {\iffieldundef{url}
        {\iffieldundef{eprint}
          {}
          {\printfield[eprint:arxiv]{eprint}}}
        {\printfield{url}}}
      {\printfield{doi}}%
    \iffieldundef{usera}{}{~\printfield{usera}}%
  }}%
}
